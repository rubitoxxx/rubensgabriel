def adicionar(display, valor):
    """Adiciona um caractere ao display, evitando erros comuns."""
    operadores = ['/', '*', '-', '+']

    # Evita múltiplos operadores em sequência
    if display and valor in operadores and display[-1] in operadores:
        return display

    # Evita adicionar um segundo ponto no número atual
    if valor == '.':
        ultimo_operador = -1
        for i, c in enumerate(display):
            if c in operadores:
                ultimo_operador = i
        if '.' in display[ultimo_operador + 1:]:
            return display

    return display + valor


def calcular(display):
    """Avalia a expressão do display e retorna o resultado."""
    try:
        resultado = eval(display)
        if isinstance(resultado, float):
            resultado = round(resultado, 4)
        return resultado
    except Exception:
        return "Erro"


def limpar():
    """Limpa o display."""
    return ""


def main():
    display = ""
    print("Calculadora em Python (digite 'q' para sair)")

    while True:
        print(f"\nDisplay: {display}")
        entrada = input("Número/operador, '=' calcular, 'c' limpar, 'q' sair: ")

        if entrada == 'q':
            break
        elif entrada == 'c':
            display = limpar()
        elif entrada == '=':
            resultado = calcular(display)
            print(f"Resultado: {resultado}")
            display = str(resultado)
        else:
            display = adicionar(display, entrada)


if __name__ == "__main__":
    main()
